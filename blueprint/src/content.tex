\chapter{Metric-Measure Volume Entropy, Bishop--Gromov Growth, and Scaling}

\begin{definition}[Logarithmic ball-volume rate, exponential growth, and entropy-volume product]
\label{def_mm_volume_entropy}
\lean{RCDEntropyDegree.MetricMeasure.ballVolumeLogRate, RCDEntropyDegree.MetricMeasure.HasExponentialVolumeGrowth, RCDEntropyDegree.MetricMeasure.normalizedExponentialWeight, RCDEntropyDegree.MetricMeasure.entropyVolumeProduct}
\leanok
For a metric-measure space $(Z, \mathsf{d}, \mathfrak{m})$ and basepoint $x_0 \in Z$, the logarithmic ball-volume rate is $R^{-1} \log \mathfrak{m}(B(x_0, R))$, exponential volume growth $\mathfrak{m}(B(x_0, R)) \le C e^{h R}$ is captured by \texttt{HasExponentialVolumeGrowth}, the normalized exponential weights $w_s(x, z)$ normalize $e^{-s\,\mathsf{d}(x, z)}$ over finite samples, and the scale-invariant entropy-volume product is $h(\overline{X})^N \mathcal{H}^N(X)$.
\end{definition}

\begin{theorem}[Basepoint independence and Bishop--Gromov hyperbolic growth bound]
\label{thm_basepoint_bishop_gromov}
\lean{RCDEntropyDegree.MetricMeasure.ball_subset_ball_add_dist, RCDEntropyDegree.MetricMeasure.measure_ball_le_measure_ball_add_dist, RCDEntropyDegree.MetricMeasure.toReal_measure_ball_le_add_dist, RCDEntropyDegree.MetricMeasure.HasExponentialVolumeGrowth.of_basepoint, RCDEntropyDegree.MetricMeasure.HasExponentialVolumeGrowth.ballVolumeLogRate_le, RCDEntropyDegree.MetricMeasure.hyperbolicSn_le_mul_exp, RCDEntropyDegree.MetricMeasure.hyperbolicDensity_le_pow_mul_exp}
\leanok
\uses{def_mm_volume_entropy}
By the triangle inclusion $B(x_0, R) \subseteq B(x_1, R + \mathsf{d}(x_0, x_1))$, exponential volume growth with exponent $h$ transfers across all basepoints $x_0, x_1 \in Z$ and bounds $R^{-1}\log\mathfrak{m}(B(x_0, R)) \le h + R^{-1}\log C$. Moreover, the hyperbolic model density satisfies $(\sinh(\kappa r)/\kappa)^{N-1} \le (2\kappa)^{-(N-1)} e^{(N-1)\kappa r}$, giving the Bishop--Gromov upper bound $h(\overline{X}) \le (N-1)\kappa$ under $\mathsf{RCD}(-(N-1)\kappa^2, N)$.
\end{theorem}
\begin{proof}\leanok Triangle inequality ball monotonicity and the elementary estimate $2\sinh(u) \le e^u$ raised to the $(N-1)$-st power.\end{proof}

\begin{theorem}[Isometric invariance and $\pi_1$-equivariance of normalized exponential weights]
\label{thm_equivariant_weights}
\lean{RCDEntropyDegree.MetricMeasure.exponentialWeight_isometry, RCDEntropyDegree.MetricMeasure.normalizedExponentialWeight_pos, RCDEntropyDegree.MetricMeasure.normalizedExponentialWeight_nonneg, RCDEntropyDegree.MetricMeasure.sum_normalizedExponentialWeight_eq_one, RCDEntropyDegree.MetricMeasure.normalizedExponentialWeight_equivariant}
\leanok
\uses{def_mm_volume_entropy}
Every distance-preserving action $\gamma$ preserves $e^{-s\,\mathsf{d}(x, z)}$ and hence yields strictly positive, probability-normalized, $\pi_1(X)$-equivariant weights $w_s(\gamma x, \gamma z_i) = w_s(x, z_i)$.
\end{theorem}
\begin{proof}\leanok Direct invariance of $\mathsf{d}(\gamma x, \gamma z) = \mathsf{d}(x, z)$ inside the numerator and denominator sum.\end{proof}

\begin{theorem}[Homothetic invariance, normalization, and finite orientation-cover reduction]
\label{thm_homothetic_cover_invariance}
\lean{RCDEntropyDegree.MetricMeasure.exp_rescaled_dist_eq, RCDEntropyDegree.MetricMeasure.logRate_homothetic_scaling, RCDEntropyDegree.MetricMeasure.entropyVolumeProduct_homothetic_invariance, RCDEntropyDegree.MetricMeasure.homothetic_normalization, RCDEntropyDegree.MetricMeasure.finite_cover_entropy_degree_reduction}
\leanok
\uses{thm_basepoint_bishop_gromov}
Under homothetic metric rescaling $(\mathsf{d}, \mathcal{H}^N) \mapsto (\lambda \mathsf{d}, \lambda^N \mathcal{H}^N)$ ($\lambda > 0$), volume entropy scales by $\lambda^{-1}$ and Hausdorff volume by $\lambda^N$, leaving $h(\overline{X})^N \mathcal{H}^N(X)$ invariant. Passing to the $k$-sheeted intermediate cover $\widehat{Y} \to Y$ associated to $f_* \pi_1(X) \le \pi_1(Y)$ with $|\deg(f)| = k |\deg(\hat{f})|$ and normalizing by $c = h(\overline{X})/h(\widetilde{Y})$ yields $|\deg(\hat{f})| = 1$, $c^{-1} h(\overline{X}) = h(\widetilde{Y})$, and $c^N \mathcal{H}^N(X) = \mathcal{H}^N(\widehat{Y})$.
\end{theorem}
\begin{proof}\leanok Cancellation of $\lambda^{-N}\lambda^N = 1$ and positive factor $k > 0$ in the covering degree factorization.\end{proof}

\chapter{Sturm's $\mathrm{CAT}(0)$ Barycenter and Equivariant Natural Maps}

\begin{definition}[Sturm's renormalized $d^2$-barycenter functional and geodesic homotopy]
\label{def_sturm_functional}
\lean{RCDEntropyDegree.MetricMeasure.sturmSummand, RCDEntropyDegree.MetricMeasure.sturmFunctional, RCDEntropyDegree.MetricMeasure.IsCATZeroMidpoint, RCDEntropyDegree.MetricMeasure.geodesicHomotopyLift, RCDEntropyDegree.MetricMeasure.geodesicHomotopyQuotient}
\leanok
For a probability weight $w$ supported at points $y_i$ in a metric space $(Y, \mathsf{d}_Y)$ and reference basepoint $o \in Y$, Sturm's renormalized $d^2$-barycenter functional is $\mathcal{S}_{w, y, o}(z) = \sum_i w_i (\mathsf{d}_Y(y_i, z)^2 - \mathsf{d}_Y(o, y_i)^2)$. Together with the $\mathrm{CAT}(0)$ semi-parallelogram midpoint predicate \texttt{IsCATZeroMidpoint}, this supports the lifted geodesic homotopy $\overline{H}_s(x, t)$ and its quotient descent $H_s : X \times [0, 1] \to Y$.
\end{definition}

\begin{lemma}[First-moment linear growth bound and basepoint independence]
\label{lem_sturm_growth_basepoint}
\lean{RCDEntropyDegree.MetricMeasure.sturmSummand_basepoint_self, RCDEntropyDegree.MetricMeasure.sturmSummand_sub_sturmSummand, RCDEntropyDegree.MetricMeasure.abs_sturmSummand_le, RCDEntropyDegree.MetricMeasure.sturmFunctional_basepoint_self, RCDEntropyDegree.MetricMeasure.sturmFunctional_eq_two_mul_metricEnergy_sub, RCDEntropyDegree.MetricMeasure.sturmFunctional_sub_basepoint, RCDEntropyDegree.MetricMeasure.sturmFunctional_min_iff_basepoint, RCDEntropyDegree.MetricMeasure.sturmFunctional_min_iff_metricEnergy, RCDEntropyDegree.MetricMeasure.continuous_sturmFunctional, RCDEntropyDegree.MetricMeasure.abs_sturmFunctional_le}
\leanok
\uses{def_sturm_functional}
Sturm's summand satisfies $|\mathsf{d}(y, z)^2 - \mathsf{d}(o, y)^2| \le \mathsf{d}(o, z)(\mathsf{d}(o, z) + 2\,\mathsf{d}(o, y))$, requiring only finite first moments, and shifting $o_1 \mapsto o_2$ adds a constant independent of $z$. Consequently, $\mathcal{S}_{w, y, o}$ is continuous and shares exact global minimizers with \texttt{metricEnergy}.
\end{lemma}
\begin{proof}\leanok Factor $a^2 - b^2 = (a - b)(a + b)$ with triangle inequality bounds and note that $\mathcal{S}_{w, y, o}(z) = 2\,\mathrm{metricEnergy}(w, y, z) - \sum_i w_i\,\mathsf{d}(o, y_i)^2$.\end{proof}

\begin{theorem}[$\mathrm{CAT}(0)$ midpoint deficit, Sturm variance inequality, and uniqueness]
\label{thm_sturm_cat0_variance}
\lean{RCDEntropyDegree.MetricMeasure.IsCATZeroMidpoint.strictMidJensenOn_halfSqDist, RCDEntropyDegree.MetricMeasure.sturmFunctional_midpoint_le, RCDEntropyDegree.MetricMeasure.sturm_variance_inequality, RCDEntropyDegree.MetricMeasure.sturmFunctional_unique_min}
\leanok
\uses{lem_sturm_growth_basepoint}
Under \texttt{IsCATZeroMidpoint}, $\mathcal{S}_{w, y, o}$ satisfies the strong midpoint convexity deficit $\mathcal{S}_{w, y, o}(m) \le \frac{1}{2}\mathcal{S}_{w, y, o}(a) + \frac{1}{2}\mathcal{S}_{w, y, o}(b) - \frac{1}{4}\mathsf{d}_Y(a, b)^2$. Any minimizer $q = \mathrm{bar}(\nu)$ therefore satisfies Sturm's variance inequality $\frac{1}{2}\mathsf{d}_Y(q, z)^2 \le \mathcal{S}_{w, y, o}(z) - \mathcal{S}_{w, y, o}(q)$ and is the unique global minimizer.
\end{theorem}
\begin{proof}\leanok Weighted summation of the $\mathrm{CAT}(0)$ semi-parallelogram law and comparison at $m = \mathrm{mid}(q, z)$ against minimality at $q$.\end{proof}

\begin{theorem}[$\pi_1$-equivariance of Sturm natural maps and geodesic homotopy descent (Lemma 3.1)]
\label{thm_sturm_equivariant_natural_map}
\lean{RCDEntropyDegree.MetricMeasure.sturmFunctional_isometry, RCDEntropyDegree.MetricMeasure.sturmBarycenter_equivariant, RCDEntropyDegree.MetricMeasure.sturmNaturalMap_equivariant, RCDEntropyDegree.MetricMeasure.geodesicHomotopyLift_zero, RCDEntropyDegree.MetricMeasure.geodesicHomotopyLift_one, RCDEntropyDegree.MetricMeasure.geodesicHomotopyLift_equivariant, RCDEntropyDegree.MetricMeasure.geodesicHomotopyQuotient_zero, RCDEntropyDegree.MetricMeasure.geodesicHomotopyQuotient_one}

For any isometric $\pi_1(X)$-representation $\rho : \Gamma \curvearrowright \widetilde{Y}$, unique minimization forces $\mathrm{bar}(\rho(\gamma)_* \nu) = \rho(\gamma)\,\mathrm{bar}(\nu)$. Combining with $\Gamma$-equivariance of $w_s(x, z)$ yields $\rho$-equivariance of the lifted natural map $\overline{F}_s$ and of the geodesic homotopy $\overline{H}_s$, which therefore descends to a homotopy $H_s : X \times [0, 1] \to Y$ between $f$ ($t = 0$) and $F_s$ ($t = 1$).
\end{theorem}
\begin{proof}\leanok Isometry covariance of $\mathcal{S}_{w, y, o}$ transfers the minimizing property of $q$ to $\rho(\gamma) q$, and uniqueness yields equality.\end{proof}

\chapter{Rectifiable Jacobians and $\mathsf{RCD}(K, N)$ Entropy-Degree Rigidity}

\begin{definition}[Metric Jacobian on Euclidean approximate tangent spaces (Lemma 3.2)]

At $\mathcal{H}^N$-almost every regular point $x \in \mathcal{R}_N(X)$ where $\mathrm{Tan}(X, x) \cong E$ ($\dim_\R E = N$), the approximate linear differential $d_x F : E \to_l[\R] E$ defines the non-negative metric Jacobian $\mathrm{Jac}_x F := |\det(d_x F)|$.
\end{definition}

\begin{lemma}[Pushforward mass-Jacobian inequality (Lemma 3.3)]
\label{lem_mass_jacobian_pushforward}

$$|\deg(f)|\,\mathcal{H}^N(Y) = \mathbf{M}(F_\# T) \le \int_Y \mathrm{mult}_F(y)\,d\mathcal{H}^N(y) = \int_X \mathrm{Jac}_x F \, d\mathcal{H}^N(x).$$
\end{lemma}
\begin{proof}\leanok Triangle inequality over finite fibers integrated against $\mathcal{H}^N_Y$, combined with Proposition 2.5.\end{proof}

\begin{theorem}[Pointwise entropy-Jacobian bound on $\mathcal{R}_N(X)$ (Proposition 3.4)]

\end{theorem}
\begin{proof}\leanok Direct instantiation of \texttt{BCG.RankOne.abs\_det\_le\_div\_entropy\_pow\_of\_rankOne\_det\_ineq}.\end{proof}

\begin{theorem}[Theorem 1.1 Part 1: Non-collapsed $\mathsf{RCD}(K, N)$ and Alexandrov entropy-degree inequality]
\label{thm_rcd_entropy_degree_inequality}
\lean{RCDEntropyDegree.RCD.rcd_entropy_degree_inequality, RCDEntropyDegree.RCD.rcd_rankOne_entropy_degree_theorem, RCDEntropyDegree.RCD.rcd_homological_index_entropy_inequality, RCDEntropyDegree.RCD.alexandrov_entropy_degree_inequality}

\begin{theorem}[Theorem 1.1 Part 2: Mass-Jacobian sandwich and equality-case rigidity]
\label{thm_rcd_equality_rigidity}
\lean{RCDEntropyDegree.RCD.ae_jacobian_eq_one_of_normalized_equality, RCDEntropyDegree.RCD.rcd_entropy_degree_equality_rigidity}
\leanok

When $\deg(f) \ne 0$, $|\deg(f)| = [\pi_1(Y) : f_* \pi_1(X)] = k \ge 1$, and equality holds in Theorem 1.1, homothetic rescaling by $c = h(\overline{X})/h(\widetilde{Y})$ and lifting to the degree-$k$ cover $p : \widehat{Y} \to Y$ give $|\deg(\hat{f})| = 1$ and $\mathcal{H}_c^N(X) = \mathcal{H}^N(\widehat{Y})$. The $1$-Lipschitz limit map $\hat{F}_\infty : X \to \widehat{Y}$ then saturates the mass-Jacobian sandwich ($\mathrm{Jac}_x \hat{F}_\infty = 1$ a.e.), is surjective, promotes to a global metric isometry $(X, c\,\mathsf{d}_X) \xrightarrow{\sim} \widehat{Y}$, and forces $\mathcal{S}(X) = X \setminus \mathcal{R}_N(X) = \emptyset$.
\end{theorem}
\begin{proof}\leanok Combines \texttt{finite\_cover\_entropy\_degree\_reduction}, \texttt{homothetic\_normalization}, the a.e.\ integral sandwich $\mathcal{H}^N(\widehat{Y}) \le \int_X \mathrm{Jac}_x \hat{F}_\infty \le \mathcal{H}_c^N(X) = \mathcal{H}^N(\widehat{Y})$, surjectivity from $\deg(\hat{F}_\infty) \ne 0$, and Lipschitz-volume rigidity.\end{proof}
